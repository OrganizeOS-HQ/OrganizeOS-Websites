// Scratch: Task 5's measure file. Never committed.
//
// Chunk mode. Each row writes its record to ./r1-m/<id>.json as it ends
// (commands.writeFile, not a snapshot, so a killed run keeps it); a row whose
// file exists is skipped, so a killed run resumes where it stopped. A run
// that freezes the page (a head defect: no timer fires, so the 20 s
// watchdog below cannot) is killed from outside by the runner,
// t5-measure-run.py, when no record has been written for 100 s; the runner
// then lists the first row without a file in ./r1-hangs.json, which records
// it as `hang`, and runs again. A row that fails (a throw, or the 20 s
// watchdog) is recorded as `failed`, and the rest of that run writes
// nothing; the runner runs again from the next row. The test "merge", run last on its own
// (`-t '^merge$'`), reads every ./r1-m file, fails on any missing row, and
// writes the one combined output: the M records (M56 and M57 included) to
// ./r1-measurements.head.json and the P records to ./r1-records.proto.json.
// At <base> the P file is the P rows' base results (r1-p-at-base.json); at
// the carry the M file is r1-measurements.proto-all.json. To record again at
// a new <base>: empty ./r1-m and ./r1-hangs.json, then run the runner.
//
// Each record with a write carries `fixedPoint`: the full written value
// (the field's export right after the action) mounted again in the field
// exports the same value (plain() on both sides). `fixedPointOut` is that
// export when it differs, `readOnly` the field's read-only reason for the
// written value. A record whose undo does not restore also carries
// `afterUndo` and `caretAfterUndo`, from a second run of the row.

const withWatchdog = <T,>(id: string, work: Promise<T>) =>
  Promise.race([
    work,
    new Promise<never>((_, reject) =>
      setTimeout(() => reject(new Error(`watchdog ${WATCHDOG} ms`)), WATCHDOG)
    ),
  ]);

type LexicalElement = HTMLElement & {
  __lexicalEditor?: ReturnType<typeof editorOf>;
};
/** The editor of the field that holds the focus, or the harness's. */
const focusedEditor = () =>
  (
    document.activeElement?.closest?.('[contenteditable="true"]') as
      | LexicalElement
      | null
      | undefined
  )?.__lexicalEditor ?? editorOf()!;

let captured: string | undefined;
const capture: Step = async () => {
  captured = exportRichTextHtml(focusedEditor());
};

/** The written value, mounted again in the field: does it export itself? */
const fixedPointOf = async (html: string) => {
  const stored = sanitizeRichTextHtml(html);
  const readOnly = getRichTextReadOnlyReason(stored);
  await mount(stored);
  await settle();
  const editor = editorOf();
  if (editor === undefined) {
    return { fixedPoint: false, readOnly: String(readOnly ?? "no editable") };
  }
  const out = exportRichTextHtml(editor);
  const fixedPoint = plain(out) === plain(stored);
  return {
    fixedPoint,
    ...(fixedPoint ? {} : { fixedPointOut: short(out) }),
    ...(readOnly === undefined ? {} : { readOnly: String(readOnly) }),
  };
};

/** A second run to the first Ctrl+Z: what the undo left. */
const undoProbe = async (r: Row) => {
  await mount(r.stored);
  await r.place?.();
  await r.action();
  await keys("{Control>}z{/Control}")();
  const editor = focusedEditor();
  return {
    afterUndo: short(exportRichTextHtml(editor)),
    caretAfterUndo: caretKey(editor),
  };
};

const record = async (r: Row) => {
  if (hangList.includes(r.id)) {
    return {
      id: r.id,
      hang: "the page froze: no record within the runner's 100 s idle watchdog",
    };
  }
  try {
    captured = undefined;
    const result = await withWatchdog(
      r.id,
      recordEdit(
        { stored: r.stored, place: r.place, action: then(r.action, capture) },
        { typeAfter: true }
      )
    );
    const clip = r.clip ? clipboard() : undefined;
    const extra: Record<string, unknown> = {};
    if (result.write !== null && captured !== undefined) {
      if (short(captured) !== result.write) {
        extra.captureDiffers = short(captured);
      }
      Object.assign(extra, await withWatchdog(r.id, fixedPointOf(captured)));
    } else if (result.write !== null) {
      extra.fixedPoint = "not captured";
    }
    if (result.undoRestores === false) {
      Object.assign(extra, await withWatchdog(r.id, undoProbe(r)));
    }
    return {
      id: r.id,
      ...result,
      ...extra,
      ...(clip === undefined ? {} : { clipboard: clip }),
    };
  } catch (error) {
    return { id: r.id, failed: String((error as Error)?.message ?? error).slice(0, 400) };
  }
};

// A failed row (a throw, or the 20 s watchdog with the action still
// running) leaves the page in no state to measure: the rows after it in
// this run write nothing, and the runner starts a new run after it.
let broken = false;
for (const r of rows) {
  if (done.has(fileOf(r.id))) {
    continue;
  }
  test(r.id, { timeout: 90000 }, async () => {
    if (broken) {
      return;

    const result = await record(r);
    if ("failed" in result) {
      broken = true;
    }
    await commands.writeFile(`./r1-m/${fileOf(r.id)}.json`, JSON.stringify(result, null, 1) + "\n");
    // D6's over-copy, a checked fact (Step 1): M20b's copy holds the next

    if ((r.id === "M20b" || r.id === "M20a") && "clipboard" in result) {
      const html = (result as { clipboard: Record<string, string> }).clipboard["text/html"] ?? "";
      if (r.id === "M20b") {
        expect(html).toContain("Iw");
      }
      expect(html).toContain("<br");
    }
  });
}

const POINTS: Array<[string, string, (n: number) => Step]> = [];
const addPoints = (shape: string, stored: string, count: number, place?: (n: number) => Step) => {
  for (let n = 0; n < count; n += 1) {
    POINTS.push([`${shape}@${n}`, stored, place ?? ((k) => itemPoint(1, k))]);
  }
};

  ["TQ@1", L("T0w" + BQ("Q1w")), itemPoint(1, 1)],
  ["QT@1", L(BQ("Q0w") + "T1w"), itemPoint(1, 1)],
  ["TQT el1", L("T0w" + BQ("Q1w") + "T2w"), itemPoint(1, 1)],

  ["QB el1", L(BQ("Q0w") + "<br>"), itemPoint(1, 1)],
  ["root el1", "<blockquote>X1w<p>X2w</p></blockquote>", containerPoint("X1w", 1)],
  ["BQ el1", L("<br>" + BQ("Q1w")), itemPoint(1, 1)],
];
test("M55", { timeout: 300000 }, async () => {
  const out: Array<{ point: string; x: number; y: number }> = [];
  const all: Array<[string, string, Step]> = [
    ...POINTS.map(([p, s, f]) => [p, s, f(Number(p.split("@")[1]))] as [string, string, Step]),
    ...MEETING,
  ];
  for (const [point, stored, place] of all) {
    await mount(stored);
    await place();
    await settle();
    const box = editable()!.getBoundingClientRect();
    const selection = window.getSelection()!;
    const sel = selection.getRangeAt(0);
    let rect: { x: number; y: number } | undefined;
    const first = sel.getClientRects()[0];
    if (first !== undefined) {
      rect = { x: first.left + first.width / 2, y: first.top + first.height / 2 };
    } else {
      const container = sel.startContainer;
      const offset = sel.startOffset;
      const rectsOf = (node: Node) => {
        const r = document.createRange();
        r.selectNode(node);
        return [...r.getClientRects()];
      };
      const before = container.childNodes[offset - 1];
      const after = container.childNodes[offset];
      if (before !== undefined) {
        const list = rectsOf(before);
        const last = list[list.length - 1];
        rect = last && { x: last.right + 1, y: last.top + last.height / 2 };
      }
      if (rect === undefined && after !== undefined) {
        const list = rectsOf(after);
        const firstRect = list[0];
        rect = firstRect && { x: firstRect.left - 1, y: firstRect.top + firstRect.height / 2 };
      }
    }
    out.push({
      point,
      x: Math.round((rect?.x ?? NaN) - box.left),
      y: Math.round((rect?.y ?? NaN) - box.top),
    });
  }
  await expect(JSON.stringify(out, null, 1) + "\n").toMatchFileSnapshot("./r1-visible-places.head.json");
});
